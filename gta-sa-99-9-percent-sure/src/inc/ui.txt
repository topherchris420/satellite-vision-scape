// ============================================================================
// RAIN_DECIDE - one R.A.I.N. validation decision.
// In: cId. Out: cChoice, cCorrect. Scores the decision and plays the reaction.
// Claims with cRetry = 1 must be answered correctly before moving on.
// ============================================================================
:RAIN_LOAD_CLAIM
{$INCLUDE ../gen/claims_data.txt}
return

:RAIN_DECIDE
gosub @RAIN_LOAD_CLAIM
uPrevMode = uMode
iSkipSaved = iSkipOk            // the confirm button must not also skip the cutscene
iSkipOk = 0
iSkipReq = 0
iTmp3 = 0                       // attempts
:RAIN_DECIDE_AGAIN
set_player_control $player1 false
uMode = 2
uSel = 2
uStatus = 0
uFeedback = 0
uStickPrev = 1                  // require the stick to return to centre first
uBtnPrev = 1
:RAIN_DECIDE_LOOP
gosub @TICK
if iFail > 0
then
    uMode = uPrevMode
    iSkipOk = iSkipSaved
    return
end
uStick = get_pad_state 0 0
if is_key_pressed 37
then
    uStick = -128
end
if is_key_pressed 39
then
    uStick = 128
end
if -60 > uStick
then
    if uStickPrev == 0
    then
        if uSel > 1
        then
            uSel -= 1
            add_one_off_sound 0.0 0.0 0.0 1028
        end
    end
    uStickPrev = 1
else
    if uStick > 60
    then
        if uStickPrev == 0
        then
            if 3 > uSel
            then
                uSel += 1
                add_one_off_sound 0.0 0.0 0.0 1028
            end
        end
        uStickPrev = 1
    else
        uStickPrev = 0
    end
end
uBtn = get_pad_state 0 15
uConfirm = get_pad_state 0 16
uBtn += uConfirm
if is_key_pressed 13
then
    uBtn = 255
end
if uBtn > 0
then
    if uBtnPrev == 0
    then
        jump @RAIN_DECIDE_CHOSEN
    end
    uBtnPrev = 1
else
    uBtnPrev = 0
end
jump @RAIN_DECIDE_LOOP

:RAIN_DECIDE_CHOSEN
clear_help
cChoice = uSel
iTmp3 += 1
// Only the first decision enters the evidence score. Corrective teaching
// retries still explain the result, but cannot earn points or inflate accuracy.
if iTmp3 == 1
then
    iDecTotal += 1
end
if cChoice == cAns
then
    cCorrect = 1
    if iTmp3 == 1
    then
        iScore += 10
        iDecRight += 1
        uFeedback = 1
    else
        uFeedback = 3
    end
    uStatus = cChoice
    add_one_off_sound 0.0 0.0 0.0 1029
else
    cCorrect = 0
    if iTmp3 == 1
    then
        iScore -= 5
        if 0 > iScore
        then
            iScore = 0
        end
        uFeedback = 2
    else
        uFeedback = 4
    end
    uStatus = 4
    add_one_off_sound 0.0 0.0 0.0 1030
end
uFlashT = iNow
uFlashT += 2600
gosub @PUBLISH
switch cChoice
    case 1
        dLine = cLineAcc
    case 2
        dLine = cLineRej
    default
        dLine = cLineAbs
end
gosub @SAY
if cCorrect == 0
then
    if cRetry == 1
    then
        if cRetryLine > 0
        then
            dLine = cRetryLine
            gosub @SAY
        end
        jump @RAIN_DECIDE_AGAIN
    end
end
iWaitMs = 900
gosub @WAIT_MS
uMode = uPrevMode
uFeedback = 0
iSkipOk = iSkipSaved
iSkipPrev = 1
if iInCs == 0
then
    set_player_control $player1 true
end
return
