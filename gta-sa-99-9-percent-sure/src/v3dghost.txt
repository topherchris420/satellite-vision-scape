then
    return
end

then
    if locate_char_on_foot_3d $scplayer fDX fDY fDZ 4.0 4.0 4.0 true
    then
        iTmp = 1
    end
end

then
    iRxOn = 1
    iSweepT = 0
    oKind = OBJ_WALK
    fOX = fDX
    fOY = fDY
    fOZ = fDZ
    jump @SWEEP_LOOP
end

then
    print_now 'VGFO5' 7000 1
    iSweepOk = 1
end

if gMeter >= 94
then
    if is_char_stopped $scplayer
    then
        iSweepT += iDt
    else
        iSweepT = 0
    end
else
    iSweepT = 0
end
if iSweepT >= SWEEP_HOLD_MS
then
    iRxOn = 1
    oKind = OBJ_NONE
    iSfxId = SFX_LOCK
    gosub @SFX_PLAY
    dLine = 302
    dQEnd = 302
