// ============================================================================
// Vers3Dynamics expansion - core systems shared by the HQ and missions 2-5
//
// Each script provides: :DLG_LOOKUP (its generated line table), :HUD_DRAW,
// :WATCH_MISSION (extra failure checks, may just return) and its own cleanup,
// which ends with gosub @CLEANUP_COMMON.
//
// Gosub depth (the engine allows 8). Scripts keep their loops at depth 1-2 and
// call helpers at depth 3, so TICK runs at 4 at most:
//   TICK(4) > DLG_UPDATE(5) > DLG_START(6) > DLG_LOOKUP / DLG_STOP(7)
//   TICK(4) > HUD_DRAW(5) > DRAW_*(6) > TS_* / helpers(7)
//   TICK(4) > WATCHDOG(5) > WATCH_MISSION(6)
// ============================================================================

:TICK
wait 0
iNow = get_game_timer
gosub @DLG_UPDATE
gosub @HUD_DRAW
gosub @WATCHDOG
if iSkipOk == 1
then
    gosub @SKIP_CHECK
end
return

:WAIT_MS
iWaitEnd = get_game_timer
iWaitEnd += iWaitMs
:WAIT_MS_LOOP
gosub @TICK
if iSkipReq == 1
then
    return
end
if iFail <> 0
then
    return
end
if iWaitEnd > iNow
then
    jump @WAIT_MS_LOOP
end
return

:CS_END

// ---------------------------------------------------------------------------
// Dialogue (same behaviour as mission 1; voice files are optional)
// ---------------------------------------------------------------------------
:SAY
dQNext = 0
if iSkipReq == 1
then
    return
end
gosub @DLG_START
while dActive == 1
    gosub @TICK
    if iSkipReq == 1
    then
        break
    end
    if iFail <> 0
    then
        break
    end
end
gosub @DLG_STOP
return

// Background conversation dLine..dQEnd (driving banter, ambient chatter).
:SAY_QUEUE
iTmp = dLine
gosub @DLG_CANCEL
dLine = iTmp
dQNext = dLine
dQNext += 1
gosub @DLG_START
return

:DLG_CANCEL
dQNext = 0
gosub @DLG_STOP
return

:DLG_START
if or
    dActive == 1
    dStream <> 0
then
    gosub @DLG_STOP
end
dHasAudio = 0
dStream = 0
gosub @DLG_LOOKUP
string_format dPath "%s.mp3" dKey
set_current_directory "cleo\99sure\audio"
if load_audio_stream dPath dStream
then
    dHasAudio = 1
end
set_current_directory 0
dLimit = dDur
if dHasAudio == 1
then
    set_audio_stream_volume dStream fVoiceVol
    set_audio_stream_state dStream 1
    dLimit += 9000
    iTmp = dLimit
else
    iTmp = dDur
end
iTmp += 500
print_now dKey iTmp 1
dT0 = get_game_timer
dActive = 1
dPed = -1
switch dWho
    case 1
        dPed = hPlayer
    case 2
        dPed = hChris
    case 3
        dPed = hNpc1
    case 4
        dPed = hNpc2
    case 5
        dPed = hNpc3
end
if dPed <> -1
then
    if not is_char_dead dPed
    then
        start_char_facial_talk dPed dLimit
        if dGest > 0
        then
            if iAnimsReady == 1
            then
                if is_char_on_foot dPed
                then
                    {$INCLUDE ../gen/v3d_gestures.txt}
                end
            end
        end
    end
end
return

:DLG_STOP
if dStream <> 0
then
    set_audio_stream_state dStream 0
    remove_audio_stream dStream
    dStream = 0
end
if dActive == 1
then
    dActive = 0
    clear_small_prints
    if dPed <> -1
    then
        if not is_char_dead dPed
        then
            stop_char_facial_talk dPed
        end
    end
end
return

:DLG_UPDATE
if dActive == 1
then
    dElapsed = iNow
    dElapsed -= dT0
    dDone = 0
    if dHasAudio == 1
    then
        dAudioState = get_audio_stream_state dStream
        if and
            dAudioState == -1
            dElapsed > 400
        then
            dDone = 1
        end
    end
    if dElapsed > dLimit
    then
        dDone = 1
    end
    if dDone == 1
    then
        gosub @DLG_STOP
        dGapT = iNow
        dGapT += 350
    end
else
    if dQNext > 0
    then
        if iNow > dGapT
        then
            if dQEnd >= dQNext
            then
                dLine = dQNext
                dQNext += 1
                gosub @DLG_START
            else
                dQNext = 0
            end
        end
    end
end
return

// ---------------------------------------------------------------------------
// Failure conditions shared by every script.
// ---------------------------------------------------------------------------
:WATCHDOG
if iFail <> 0
then
    return
end
if iNeedChris > 0
then
    if is_char_dead hChris
    then
        iFail = 1
        return
    end
end
if iNeedVan == 1
then
    if is_car_dead hVan
    then
        iFail = 2
        return
    end
end
if iNeedChris == 2
then
    if locate_char_any_means_char_2d $scplayer hChris ABANDON_WARN ABANDON_WARN false
    then
        if bChris <> -1
        then
            remove_blip bChris
            bChris = -1
        end
    else
        if bChris == -1
        then
            bChris = add_blip_for_char hChris
            set_blip_as_friendly bChris true
            print_now 'VXO01' 6000 1
        end
        if not locate_char_any_means_char_2d $scplayer hChris ABANDON_FAIL ABANDON_FAIL false
        then
            iFail = 3
            return
        end
    end
end
gosub @WATCH_MISSION
return

    rVisits = 0

// rAudio: four 4-bit levels (0..10). Level -> volume = level / 10.
:AUDIO_DECODE
iTmp = rAudio
iTmp2 = iTmp
iTmp2 = bit_and iTmp2 15
gosub @AUDIO_CLAMP
fMusicVol =# iTmp2
fMusicVol /= 10.0
iTmp2 = iTmp
iTmp2 /= 16
iTmp2 = bit_and iTmp2 15
gosub @AUDIO_CLAMP
fAmbVol =# iTmp2
fAmbVol /= 10.0
iTmp2 = iTmp
iTmp2 /= 256
iTmp2 = bit_and iTmp2 15
gosub @AUDIO_CLAMP
fVoiceVol =# iTmp2
fVoiceVol /= 10.0
iTmp2 = iTmp
iTmp2 /= 4096
iTmp2 = bit_and iTmp2 15
gosub @AUDIO_CLAMP
fSfxVol =# iTmp2
fSfxVol /= 10.0
return

:AUDIO_CLAMP

    iTmp2 = 10
end

// ---------------------------------------------------------------------------
// Menu input with edge detection: uIn = 0 none, 1 up, 2 down, 3 left, 4 right,
// 5 confirm (Enter/Exit or Enter), 6 back (Conversation - No, Backspace),
// 7 Conversation - Yes. A held button or stick counts once.
// ---------------------------------------------------------------------------
:MENU_INPUT

then
    uStick = -128
end
if is_key_pressed KEY_DOWN
then
    uStick = 128
end
uI = get_pad_state 0 PAD_STICK_X
if is_key_pressed KEY_LEFT
then
    uI = -128
end
if is_key_pressed KEY_RIGHT
then
    uI = 128
end
uI2 = 0
if -60 > uStick
then
    uI2 = 1
end
if uStick > 60
then
    uI2 = 2
end
if -60 > uI
then
    uI2 = 3
end
if uI > 60
then
    uI2 = 4
end
if uI2 > 0
then
    if uStickPrev == 0
    then
        uIn = uI2
    end
    uStickPrev = 1
else
    uStickPrev = 0
end
uBtn = get_pad_state 0 PAD_TRIANGLE
if is_key_pressed KEY_ENTER
then
    uBtn = 255
end
if uBtn > 0
then
    if uBtnPrev == 0
    then
        uIn = 5
    end
    uBtnPrev = 1
else
    uBtnPrev = 0
end
uBtn = get_pad_state 0 PAD_DPAD_LEFT
if is_key_pressed KEY_BACKSPACE
then
    uBtn = 255
end
if uBtn > 0
then
    if uBackPrev == 0
    then
        uIn = 6
    end
    uBackPrev = 1
else
    uBackPrev = 0
end
uBtn = get_pad_state 0 PAD_DPAD_RIGHT
if uBtn > 0
then
    if uYesPrev == 0
    then
        uIn = 7
    end
    uYesPrev = 1
else
    uYesPrev = 0
end

// ---------------------------------------------------------------------------
// The HQ door outside (pavement node) and a kerbside parking spot.
// ---------------------------------------------------------------------------
:FIND_LAB_EXT
fLabX, fLabY, fLabZ = get_closest_char_node LAB_REF_X LAB_REF_Y LAB_REF_Z
if fLabX == 0.0
then
    fLabX = LAB_REF_X
    fLabY = LAB_REF_Y
    fLabZ = LAB_REF_Z
end
fLabZ += 1.0
fVanX, fVanY, fVanZ, fVanH = get_closest_car_node_with_heading fLabX fLabY fLabZ
if fVanX == 0.0
then
    fVanX = fLabX
    fVanX += 6.0
    fVanY = fLabY
    fVanZ = fLabZ
    fVanH = 0.0
else
    fTmp = fLabX
    fTmp -= fVanX
    fTmp2 = fLabY
    fTmp2 -= fVanY
    fDist = get_distance_between_coords_2d fLabX fLabY fVanX fVanY
    if fDist > 2.5
    then
        fTmp /= fDist
        fTmp2 /= fDist
        fTmp *= 1.8
        fTmp2 *= 1.8
        fVanX += fTmp
        fVanY += fTmp2
    end
end
iLabSet = 1
return
